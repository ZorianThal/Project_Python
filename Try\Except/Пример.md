def divide(a, b):
    if b == 0:
        raise ValueError("Делитель не может быть равен нулю")
    return a/b

try:
    print(divide(10, 0))
except ValueError as e:
    print(e)

class AgeError(Exception):
    pass

def check_age(age):
    if age < 0:
        raise AgeError("Бро ты еще не родился")
    if age > 110:
        raise AgeError("ТЫ СТАРЫЙ")
    return "Возраст подходит"

try:
    print(check_age(int(input("введите возраст: "))))
except AgeError as e:
    print(e)
